public class Solution {
    public string GcdOfStrings(string str1, string str2) 
    {
        string result = "";
        if(str1 + str2 != str2 + str1)
        {
            return "";
        }
        int size1 = 0 , size2 = 0;
        foreach(char c in str1)
        {
            size1++;
        }
        foreach(char c in str2)
        {
            size2++;
        }
        int GCD = gcd(size1 , size2);
        for(int i = 0 ; i < GCD ; i++)
        {
            result += str1[i];
        }
        return result;
    }
    public int gcd(int a , int b)
    {
        int n = a;
        if( a > b)
        {
            n = b;
        }
        while(true)
        {
            if( a % n == 0 && b % n == 0)
            {
                return n;
            }
            n--;
        }
        return 0;
    }

}
