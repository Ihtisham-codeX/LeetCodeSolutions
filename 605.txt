public class Solution {
    public bool CanPlaceFlowers(int[] flowerbed, int n) {
        if(n == 0) return true;
        int size = flowerbed.Length;
        int count = n;
        for(int i = 0 ; i < size ; i++)
        {
            if(flowerbed[i] == 0)
            {
                if( i-1 >= 0) 
                {
                    if(flowerbed[i-1] == 0)
                    {
                        if(i+1 < size)
                        {
                            if(flowerbed[i+1] == 0)
                            {
                                flowerbed[i] =1;
                                count--;
                            }
                        }
                        else
                        {
                            flowerbed[i] =1;
                            count--;
                        }
                    }
                }
                else
                {
                    if(i+1 < size)
                        {
                            if(flowerbed[i+1] == 0)
                            {
                                flowerbed[i] =1;
                                count--;
                            }
                        }
                    else
                    {
                        if(flowerbed[i] == 0)
                        {
                            flowerbed[i] = 1;
                            count--;
                        }
                    }
                }
            }
            if(count == 0)
            {
                return true;
            }
        }
        return false;
    }
}
